-- API: Send, Success, Error, Warning, Info, ClearAll

local TweenService = game:GetService("TweenService")
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

-- ============================================================
-- Setup ScreenGui
-- ============================================================
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "NotifyLib_GUI"
ScreenGui.ResetOnSpawn = false
ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
ScreenGui.DisplayOrder = 999
ScreenGui.IgnoreGuiInset = true

-- หา parent ที่เหมาะสม (executor / CoreGui fallback)
local ok = pcall(function()
	ScreenGui.Parent = game:GetService("CoreGui")
end)
if not ok then
	ScreenGui.Parent = PlayerGui
end

-- Container วาง notify ที่มุมขวาบน
local Container = Instance.new("Frame")
Container.Name = "Container"
Container.BackgroundTransparency = 1
Container.Size = UDim2.new(0, 280, 1, 0)
Container.Position = UDim2.new(1, -295, 0, 0)
Container.Parent = ScreenGui

local ListLayout = Instance.new("UIListLayout")
ListLayout.SortOrder = Enum.SortOrder.LayoutOrder
ListLayout.Padding = UDim.new(0, 8)
ListLayout.VerticalAlignment = Enum.VerticalAlignment.Top
ListLayout.HorizontalAlignment = Enum.HorizontalAlignment.Center
ListLayout.Parent = Container

local ContainerPadding = Instance.new("UIPadding")
ContainerPadding.PaddingTop = UDim.new(0, 12)
ContainerPadding.Parent = Container

-- ============================================================
-- สี ตาม type
-- ============================================================
local COLORS = {
	success = Color3.fromRGB(39, 174, 96),
	error   = Color3.fromRGB(231, 76, 60),
	warning = Color3.fromRGB(243, 156, 18),
	info    = Color3.fromRGB(52, 152, 219),
}

local ICONS = {
	success = "✔",
	error   = "✘",
	warning = "⚠",
	info    = "ℹ",
}

-- ============================================================
-- NotifyLib
-- ============================================================
local NotifyLib = {}
local activeNotifs = {}
local layoutOrder = 0

function NotifyLib:Send(title, message, notifType, duration)
	notifType = (notifType or "info"):lower()
	duration  = duration or 5

	local accentColor = COLORS[notifType] or COLORS.info
	local icon        = ICONS[notifType]  or "ℹ"

	layoutOrder = layoutOrder + 1

	-- ── Frame หลัก ──────────────────────────────────────────
	local Card = Instance.new("Frame")
	Card.Name = "Notify_" .. layoutOrder
	Card.Size = UDim2.new(1, 0, 0, 64)
	Card.BackgroundColor3 = Color3.fromRGB(22, 22, 30)
	Card.BorderSizePixel = 0
	Card.ClipsDescendants = true
	Card.LayoutOrder = layoutOrder
	Card.Position = UDim2.new(1.2, 0, 0, 0) -- เริ่มนอกจอ
	Card.Parent = Container

	local CardCorner = Instance.new("UICorner")
	CardCorner.CornerRadius = UDim.new(0, 8)
	CardCorner.Parent = Card

	-- เส้นซ้าย (accent)
	local Accent = Instance.new("Frame")
	Accent.Size = UDim2.new(0, 4, 1, 0)
	Accent.Position = UDim2.new(0, 0, 0, 0)
	Accent.BackgroundColor3 = accentColor
	Accent.BorderSizePixel = 0
	Accent.Parent = Card

	local AccentCorner = Instance.new("UICorner")
	AccentCorner.CornerRadius = UDim.new(0, 8)
	AccentCorner.Parent = Accent

	-- icon
	local IconLabel = Instance.new("TextLabel")
	IconLabel.Size = UDim2.new(0, 30, 1, 0)
	IconLabel.Position = UDim2.new(0, 10, 0, 0)
	IconLabel.BackgroundTransparency = 1
	IconLabel.Text = icon
	IconLabel.TextColor3 = accentColor
	IconLabel.TextSize = 20
	IconLabel.Font = Enum.Font.GothamBold
	IconLabel.TextXAlignment = Enum.TextXAlignment.Center
	IconLabel.Parent = Card

	-- title
	local TitleLabel = Instance.new("TextLabel")
	TitleLabel.Size = UDim2.new(1, -50, 0, 22)
	TitleLabel.Position = UDim2.new(0, 46, 0, 8)
	TitleLabel.BackgroundTransparency = 1
	TitleLabel.Text = tostring(title)
	TitleLabel.TextColor3 = Color3.fromRGB(240, 240, 240)
	TitleLabel.TextSize = 14
	TitleLabel.Font = Enum.Font.GothamBold
	TitleLabel.TextXAlignment = Enum.TextXAlignment.Left
	TitleLabel.TextTruncate = Enum.TextTruncate.AtEnd
	TitleLabel.Parent = Card

	-- message
	local MsgLabel = Instance.new("TextLabel")
	MsgLabel.Size = UDim2.new(1, -50, 0, 20)
	MsgLabel.Position = UDim2.new(0, 46, 0, 32)
	MsgLabel.BackgroundTransparency = 1
	MsgLabel.Text = tostring(message)
	MsgLabel.TextColor3 = Color3.fromRGB(170, 170, 190)
	MsgLabel.TextSize = 12
	MsgLabel.Font = Enum.Font.Gotham
	MsgLabel.TextXAlignment = Enum.TextXAlignment.Left
	MsgLabel.TextTruncate = Enum.TextTruncate.AtEnd
	MsgLabel.Parent = Card

	-- progress bar
	local ProgressBG = Instance.new("Frame")
	ProgressBG.Size = UDim2.new(1, 0, 0, 3)
	ProgressBG.Position = UDim2.new(0, 0, 1, -3)
	ProgressBG.BackgroundColor3 = Color3.fromRGB(40, 40, 55)
	ProgressBG.BorderSizePixel = 0
	ProgressBG.Parent = Card

	local ProgressBar = Instance.new("Frame")
	ProgressBar.Size = UDim2.new(1, 0, 1, 0)
	ProgressBar.BackgroundColor3 = accentColor
	ProgressBar.BorderSizePixel = 0
	ProgressBar.Parent = ProgressBG

	-- ── Slide-in ────────────────────────────────────────────
	local tweenIn = TweenService:Create(Card,
		TweenInfo.new(0.35, Enum.EasingStyle.Quint, Enum.EasingDirection.Out),
		{ Position = UDim2.new(0, 0, 0, 0) }
	)
	tweenIn:Play()

	-- progress bar shrink
	local tweenProg = TweenService:Create(ProgressBar,
		TweenInfo.new(duration, Enum.EasingStyle.Linear),
		{ Size = UDim2.new(0, 0, 1, 0) }
	)
	tweenIn.Completed:Connect(function()
		tweenProg:Play()
	end)

	-- ── Remove ──────────────────────────────────────────────
	local function removeCard()
		if not Card or not Card.Parent then return end
		local tweenOut = TweenService:Create(Card,
			TweenInfo.new(0.3, Enum.EasingStyle.Quint, Enum.EasingDirection.In),
			{ Position = UDim2.new(1.2, 0, 0, 0) }
		)
		tweenOut:Play()
		tweenOut.Completed:Connect(function()
			Card:Destroy()
		end)
		-- ลบออกจาก activeNotifs
		for i, v in ipairs(activeNotifs) do
			if v == Card then table.remove(activeNotifs, i) break end
		end
	end

	tweenProg.Completed:Connect(removeCard)
	table.insert(activeNotifs, Card)

	return removeCard -- คืน function สำหรับ dismiss manual
end

-- ── Shorthand ────────────────────────────────────────────────
function NotifyLib:Success(title, message, duration)
	return self:Send(title, message, "success", duration)
end

function NotifyLib:Error(title, message, duration)
	return self:Send(title, message, "error", duration)
end

function NotifyLib:Warning(title, message, duration)
	return self:Send(title, message, "warning", duration)
end

function NotifyLib:Info(title, message, duration)
	return self:Send(title, message, "info", duration)
end

-- ── ClearAll ─────────────────────────────────────────────────
function NotifyLib:ClearAll()
	for _, card in ipairs(activeNotifs) do
		if card and card.Parent then
			card:Destroy()
		end
	end
	activeNotifs = {}
end

return NotifyLib
